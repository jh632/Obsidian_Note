# 环形缓冲区

## 定义

环形缓冲区（Ring Buffer / Circular Buffer）是一种固定大小的缓冲区，数据在其中循环存储。当写入到达末尾时会自动回到开头继续写入，形成一个逻辑上的环形结构。它是嵌入式系统中最常用的数据结构之一。

## 核心思想

**空间复用**：在固定内存空间中实现数据的持续流动，无需移动数据。

## 解决问题

主要解决不定长数据,解决了突发的数据速度波动

## 工作原理

```
┌─────────────────────────────────────────────────────────┐
│                    环形缓冲区结构                        │
├─────────────────────────────────────────────────────────┤
│                                                         │
│    写入方向 →                                          │
│    ┌───┬───┬───┬───┬───┬───┬───┬───┐                │
│    │ 0 │ 1 │ 2 │ 3 │ 4 │ 5 │ 6 │ 7 │  (size=8)      │
│    └───┴───┴───┴───┴───┴───┴───┴───┘                │
│      ↑                           ↑                    │
│     head                       tail                   │
│    (读位置)                    (写位置)                │
│                                                         │
│    数据方向：tail → ... → head (FIFO)                  │
└─────────────────────────────────────────────────────────┘
```

### 关键概念

- **head（头指针）**：读取位置，指向下一个要读取的数据
- **tail（尾指针）**：写入位置，指向下一个要写入的位置
- **FIFO（先进先出）**：先写入的数据先读出

## 数据结构设计

### 方式一：使用计数器（最常用）

```c
typedef struct {
    volatile uint16_t head;      // 读位置
    volatile uint16_t tail;      // 写位置
    volatile uint16_t count;     // 当前数据量
    uint16_t size;               // 缓冲区大小
    uint8_t *buf;                // 缓冲区指针
} ring_buf_t;

// 初始化
void ring_buf_init(ring_buf_t *r, uint8_t *buf, uint16_t size) {
    r->head = 0;
    r->tail = 0;
    r->count = 0;
    r->size = size;
    r->buf = buf;
}

// 写入一个字节
int ring_buf_put(ring_buf_t *r, uint8_t data) {
    if (r->count >= r->size) {
        return -1;  // 缓冲区满
    }
    r->buf[r->tail] = data;
    r->tail = (r->tail + 1) % r->size;
    r->count++;
    return 0;
}

// 读取一个字节
int ring_buf_get(ring_buf_t *r, uint8_t *data) {
    if (r->count == 0) {
        return -1;  // 缓冲区空
    }
    *data = r->buf[r->head];
    r->head = (r->head + 1) % r->size;
    r->count--;
    return 0;
}

// 获取可读数据量
uint16_t ring_buf_count(ring_buf_t *r) {
    return r->count;
}

// 判断是否为空
int ring_buf_empty(ring_buf_t *r) {
    return r->count == 0;
}

// 判断是否为满
int ring_buf_full(ring_buf_t *r) {
    return r->count >= r->size;
}
```

### 方式二：使用 head/tail 差值判断满（节省空间）

```c
typedef struct {
    volatile uint16_t head;
    volatile uint16_t tail;
    uint16_t size;              // 必须是 2 的幂
    uint8_t *buf;
} ring_buf_t;

// 写入
int ring_buf_put(ring_buf_t *r, uint8_t data) {
    uint16_t next = (r->tail + 1) & (r->size - 1);  // 取模优化
    if (next == r->head) {
        return -1;  // 满
    }
    r->buf[r->tail] = data;
    r->tail = next;
    return 0;
}

// 读取
int ring_buf_get(ring_buf_t *r, uint8_t *data) {
    if (r->head == r->tail) {
        return -1;  // 空
    }
    *data = r->buf[r->head];
    r->head = (r->head + 1) & (r->size - 1);
    return 0;
}

// 可用数据量（支持绕回）
uint16_t ring_buf_count(ring_buf_t *r) {
    return (r->tail - r->head) & (r->size - 1);
}
```

> **注意**：此方式缓冲区大小必须是 2 的幂，用 `& (size-1)` 代替 `% size` 提高效率。

### 方式三：面向对象风格（ops 模式）

```c
// ring_buf.h
typedef struct ring_buf ring_buf_t;

typedef struct {
    int (*init)(ring_buf_t *r, uint8_t *buf, uint16_t size);
    int (*put)(ring_buf_t *r, uint8_t data);
    int (*get)(ring_buf_t *r, uint8_t *data);
    int (*put_buf)(ring_buf_t *r, const uint8_t *data, uint16_t len);
    int (*get_buf)(ring_buf_t *r, uint8_t *data, uint16_t len);
    uint16_t (*count)(ring_buf_t *r);
    int (*empty)(ring_buf_t *r);
    int (*full)(ring_buf_t *r);
    void (*flush)(ring_buf_t *r);
} ring_buf_ops_t;

struct ring_buf {
    volatile uint16_t head;
    volatile uint16_t tail;
    volatile uint16_t count;
    uint16_t size;
    uint8_t *buf;
    const ring_buf_ops_t *ops;
};

// 使用
extern const ring_buf_ops_t ring_buf_ops;
```

## 批量读写

```c
// 批量写入
int ring_buf_put_buf(ring_buf_t *r, const uint8_t *data, uint16_t len) {
    if (r->count + len > r->size) {
        return -1;  // 空间不足
    }

    for (uint16_t i = 0; i < len; i++) {
        r->buf[r->tail] = data[i];
        r->tail = (r->tail + 1) % r->size;
    }
    r->count += len;
    return len;
}

// 批量读取
int ring_buf_get_buf(ring_buf_t *r, uint8_t *data, uint16_t len) {
    uint16_t avail = r->count;
    uint16_t read_len = (len > avail) ? avail : len;

    for (uint16_t i = 0; i < read_len; i++) {
        data[i] = r->buf[r->head];
        r->head = (r->head + 1) % r->size;
    }
    r->count -= read_len;
    return read_len;
}
```

## 嵌入式应用场景

### 1. UART 串口接收

```c
#define UART_RX_BUF_SIZE 256

static uint8_t uart_rx_buf[UART_RX_BUF_SIZE];
static ring_buf_t uart_rx_ring;

void UART_Init(void) {
    ring_buf_init(&uart_rx_ring, uart_rx_buf, UART_RX_BUF_SIZE);
}

// 中断接收回调
void UART_Rx_IRQHandler(uint8_t data) {
    ring_buf_put(&uart_rx_ring, data);
}

// 主循环读取处理
void UART_Process(void) {
    uint8_t ch;
    while (ring_buf_get(&uart_rx_ring, &ch) == 0) {
        // 处理接收到的数据
        process_char(ch);
    }
}
```

### 3. 日志系统

```c
#define LOG_BUF_SIZE 4096

static uint8_t log_buf[LOG_BUF_SIZE];
static ring_buf_t log_ring;

void log_init(void) {
    ring_buf_init(&log_ring, log_buf, LOG_BUF_SIZE);
}

void log_printf(const char *fmt, ...) {
    char tmp[128];
    va_list args;
    va_start(args, fmt);
    vsnprintf(tmp, sizeof(tmp), fmt, args);
    va_end(args);

    // 写入环形缓冲区
    for (int i = 0; tmp[i] != '\0'; i++) {
        ring_buf_put(&log_ring, tmp[i]);
    }
}

// 后台任务定期刷新日志
void log_flush_task(void) {
    uint8_t ch;
    while (ring_buf_get(&log_ring, &ch) == 0) {
        // 输出到串口或存储
        UART_SendByte(ch);
    }
}
```

## 中断安全处理

### 单生产者单消费者（无锁）

```c
// 生产者（中断写入）
void producer_isr(ring_buf_t *r, uint8_t data) {
    r->buf[r->tail] = data;
    r->tail = (r->tail + 1) % r->size;
    // 不需要修改 count，生产者只写 tail
}

// 消费者（主循环读取）
int consumer_poll(ring_buf_t *r, uint8_t *data) {
    if (r->head == r->tail) {
        return -1;  // 空
    }
    *data = r->buf[r->head];
    r->head = (r->head + 1) % r->size;
    return 0;
}
```

> **原理**：只修改 head 或 tail，互斥访问。无需关中断。

### 多生产者多消费者（关中断）

```c
// 关中断保护
void ring_buf_put_irq(ring_buf_t *r, uint8_t data) {
    uint32_t primask = __get_PRIMASK();
    __disable_irq();

    if (r->count < r->size) {
        r->buf[r->tail] = data;
        r->tail = (r->tail + 1) % r->size;
        r->count++;
    }

    __set_PRIMASK(primask);
}

int ring_buf_get_irq(ring_buf_t *r, uint8_t *data) {
    uint32_t primask = __get_PRIMASK();
    __disable_irq();

    int ret = -1;
    if (r->count > 0) {
        *data = r->buf[r->head];
        r->head = (r->head + 1) % r->size;
        r->count--;
        ret = 0;
    }

    __set_PRIMASK(primask);
    return ret;
}
```

### 使用 __atomic 操作

```c
// GCC 原子操作
void ring_buf_put_atomic(ring_buf_t *r, uint8_t data) {
    uint16_t tail = __atomic_load_n(&r->tail, __ATOMIC_RELAXED);
    r->buf[tail] = data;
    __atomic_store_n(&r->tail, (tail + 1) % r->size, __ATOMIC_RELEASE);
}
```

## 优缺点

### 优点

| 优点 | 说明 |
|------|------|
| **内存效率** | 固定大小，无内存泄漏 |
| **FIFO 特性** | 保证数据顺序 |
| **实时性好** | 写入/读取 O(1) 复杂度 |
| **无锁设计** | 单生产者单消费者可无锁 |
| **DMA 友好** | 适合连续内存传输 |

### 缺点

| 缺点 | 说明 |
|------|------|
| **大小固定** | 无法动态扩容 |
| **满时丢弃** | 需要策略处理（丢弃/覆盖/等待） |
| **内存碎片** | 预分配可能浪费 |
| **绕回开销** | 取模运算有少量开销 |

## 满时处理策略

```c
// 策略1：丢弃新数据
int ring_buf_put_drop(ring_buf_t *r, uint8_t data) {
    if (r->count >= r->size) {
        return -1;  // 丢弃
    }
    return ring_buf_put(r, data);
}

// 策略2：覆盖旧数据
void ring_buf_put_overwrite(ring_buf_t *r, uint8_t data) {
    r->buf[r->tail] = data;
    r->tail = (r->tail + 1) % r->size;
    if (r->count >= r->size) {
        r->head = (r->head + 1) % r->size;  // 覆盖最旧数据
    } else {
        r->count++;
    }
}

// 策略3：阻塞等待（RTOS 场景）
void ring_buf_put_block(ring_buf_t *r, uint8_t data) {
    while (ring_buf_full(r)) {
        vTaskDelay(1);  // 等待空间
    }
    ring_buf_put(r, data);
}
```

## 性能优化技巧

### 1. 取模优化

```c
// 原始取模（较慢）
index = (index + 1) % size;

// 位运算优化（size 必须是 2 的幂）
index = (index + 1) & (size - 1);
```

### 2. 预计算批量操作

```c
// 一次计算可连续写入的长度
uint16_t ring_buf_linear_write(ring_buf_t *r, uint8_t **ptr, uint16_t *len) {
    uint16_t avail = r->size - r->count;
    uint16_t to_end = r->size - r->tail;

    *len = (avail < to_end) ? avail : to_end;
    *ptr = &r->buf[r->tail];
    return *len;
}

// 更新 tail（无需循环）
void ring_buf_commit_write(ring_buf_t *r, uint16_t len) {
    r->tail = (r->tail + len) % r->size;
    r->count += len;
}
```

### 3. 峰值检测

```c
typedef struct {
    ring_buf_t ring;
    uint16_t peak_count;      // 峰值数据量
    uint32_t overflow_count;  // 溢出次数
} ring_buf_monitor_t;

int ring_buf_put_monitored(ring_buf_monitor_t *m, uint8_t data) {
    int ret = ring_buf_put(&m->ring, data);
    if (ret == 0) {
        uint16_t count = m->ring.count;
        if (count > m->peak_count) {
            m->peak_count = count;
        }
    } else {
        m->overflow_count++;
    }
    return ret;
}
```

## 与 FreeRTOS 集成

```c

#include "stream_buffer.h"

#define STREAM_BUF_SIZE 256

static StreamBufferHandle_t stream_buf;

void stream_init(void) {
    stream_buf = xStreamBufferCreate(STREAM_BUF_SIZE, 1);  // 触发字节数=1

// ISR 中写入
void UART_Rx_IRQHandler(uint8_t data) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xStreamBufferSendFromISR(stream_buf, &data, 1, &xHigherPriorityTaskWoken);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);

// 任务中读取
void uart_process_task(void *param) {
    uint8_t ch;
    while (1) {
        xStreamBufferReceive(stream_buf, &ch, 1, portMAX_DELAY);
        process_char(ch);
    }
}
```

## 典型应用场景总结

1. **UART/CAN 接收缓冲**：中断接收，主循环处理
2. **DMA 连续采集**：ADC、I2S 等数据缓冲
3. **日志系统**：异步日志写入
4. **按键/事件队列**：中断产生，主循环消费
5. **音频播放**：I2S/PCM 数据缓冲
6. **传感器数据流**：连续采样数据暂存

## 选型参考

| 场景 | 推荐大小 | 注意事项 |
|------|----------|----------|
| UART 接收 | 128~1024 字节 | 考虑波特率和处理时间 |
| ADC DMA | 采样点数 × 2 | 双缓冲 + 环形 |
| 日志 | 2~16KB | 考虑日志输出速度 |
| 音频 | 1~8KB | 考虑采样率和位宽 |
| 按键事件 | 8~32 字节 | 通常足够小 |

## 扩展：线程安全的环形缓冲区

```c
// 使用自旋锁（RTOS）
#include "FreeRTOS.h"
#include "semphr.h"

typedef struct {
    volatile uint16_t head;
    volatile uint16_t tail;
    volatile uint16_t count;
    uint16_t size;
    uint8_t *buf;
    SemaphoreHandle_t mutex;
} thread_safe_ring_t;

int ts_ring_put(thread_safe_ring_t *r, uint8_t data) {
    if (xSemaphoreTake(r->mutex, portMAX_DELAY) == pdTRUE) {
        if (r->count >= r->size) {
            xSemaphoreGive(r->mutex);
            return -1;
        }
        r->buf[r->tail] = data;
        r->tail = (r->tail + 1) % r->size;
        r->count++;
        xSemaphoreGive(r->mutex);
        return 0;
    }
    return -1;
}
```
